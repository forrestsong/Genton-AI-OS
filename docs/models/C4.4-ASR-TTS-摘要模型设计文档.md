# C4.4 本地 ASR/TTS + 摘要模型设计文档

## 任务信息

- **任务编号**: C4.4
- **任务名称**: 本地 ASR/TTS（CallGate 留言转写与代接）+ TOP_UI_SUMMARY/简报摘要小模型
- **前置依赖**: E4.1 录音与通话合规评审 ⏳
- **里程碑**: A4（主动服务与沙箱）
- **配套文档**: D16 CallGate 服务、D8 上下文服务、D9 智能小模型族

## 核心目标

实现端侧音频和文本处理能力，支持：

1. ✅ **ASR（语音识别）**: CallGate 留言转写、备忘收声
2. ✅ **TTS（语音合成）**: CallGate 代接回复
3. ✅ **TOP_UI_SUMMARY**: 上下文简报摘要
4. ✅ **性能约束**: 模型大小 ≤50MB、推理延迟 P95 ≤200ms、内存占用 ≤100MB

## 核心组件设计

### 1. ASR（自动语音识别）引擎

#### 1.1 模型选型

**候选方案**:

| 方案 | 模型 | 参数量 | 大小 | 延迟（10s 音频） | 优劣 |
|------|------|--------|------|------------------|------|
| **方案 A** | Whisper Tiny | 39M | 39MB | ~1.5s | ✅ 推荐：平衡性能与准确率 |
| **方案 B** | Whisper Base | 74M | 74MB | ~2.5s | 可选：更高准确率 |
| **方案 C** | WeNet | 45M | 45MB | ~1.2s | 备选：中文优化 |

**最终选型**: **Whisper Tiny**

**理由**:
- OpenAI 官方模型，多语言支持
- 39M 参数，满足 ≤50MB 约束
- TFLite/ONNX 量化后 ~25MB
- 中英文识别准确率 >90%
- 社区生态成熟，易于优化

#### 1.2 技术规格

```yaml
模型配置:
  架构: Whisper Tiny (Encoder-Decoder Transformer)
  输入: 
    - 采样率: 16kHz

  输出:
    - 文本（带标点）
    - 置信度分数
    - 时间戳（可选）
  
量化策略:
  - INT8 动态量化
  - 模型大小: ~25MB
  - 精度损失: <2%

性能指标:
  - 10s 音频转写延迟: ≤1.5s (P50), ≤2s (P95)
  - WER（词错误率）: ≤8% (中文), ≤5% (英文)
  - 内存占用: ≤80MB
```

```kotlin
class SummaryService(private val context: Context) {
    
    private val summaryEngine: SummaryInferenceEngine by lazy {
        SummaryInferenceEngine(context, modelPath = "t5_small_int8.tflite")
    }
    
    private val piiMaskService: PiiMaskService by lazy {
        PiiMaskService.getInstance(context)
    }
    
    /**
     * 生成摘要
     */
    fun summarize(
        text: String,
        maxLength: Int = 128,
        minLength: Int = 64,
        maskPii: Boolean = true
    ): String {
        // 1. PII 掩码（可选）
        val processedText = if (maskPii) {
            piiMaskService.maskPii(text).maskedText
        } else {
            text
        }
        
        // 2. 长度检查
        require(processedText.length <= 8192) { "Text too long (max 8192 chars)" }
        
        // 3. 生成摘要
        return summaryEngine.summarize(processedText, maxLength, minLength)
    }
    
    /**
     * 为 TOP_UI_SUMMARY 生成上下文简报
     */
    fun generateUiSummary(context: ContextFrame): String {
        val fullText = buildContextText(context)
        
        return summarize(
            text = fullText,
            maxLength = 128,
            minLength = 64,
            maskPii = true  // 强制掩码
        )
    }
    
    /**
     * 批量摘要（通知聚合）
     */
    fun summarizeBatch(texts: List<String>): List<String> {
        return texts.map { summarize(it) }
    }
    
    private fun buildContextText(context: ContextFrame): String {
        return buildString {
            append("应用: ${context.activeApp}\n")
            append("窗口: ${context.windowTitle}\n")
            if (context.textContent.isNotEmpty()) {
                append("内容: ${context.textContent}\n")
            }
            // 限制长度
        }.take(2048)
    }
}
```

```kotlin
class TtsInferenceEngine(
    context: Context,
    modelPath: String
) {
    private val ortSession: OrtSession
    
    init {
        val ortEnv = OrtEnvironment.getEnvironment()
        ortSession = ortEnv.createSession(
            loadModelBytes(context, modelPath),
            OrtSession.SessionOptions()
        )
    }
    
    fun synthesize(
        text: String,
        speed: Float,
        voiceId: String
    ): AudioStream {
        // 1. 文本预处理（分词、音素转换）
        val phonemes = textToPhonemes(text)
        
        // 2. ONNX 推理
        val inputs = OnnxTensor.createTensor(ortEnv, phonemes)
        val outputs = ortSession.run(mapOf("input" to inputs))
        
        // 3. 解码为音频
        val audioTensor = outputs[0].value as FloatArray
        val audioData = decodeAudio(audioTensor, speed)
        
        return AudioStream(audioData, sampleRate = 22050)
    }
    
    fun synthesizeStreaming(
        text: String,
        callback: (AudioChunk) -> Unit
    ) {
        // 流式生成：逐句合成，逐块回调
        val sentences = splitSentences(text)
        sentences.forEach { sentence ->
            val audio = synthesize(sentence, 1.0f, "neutral")
            callback(AudioChunk(audio.data))
        }
    }
}
```

#### SummaryInferenceEngine

```kotlin
class SummaryInferenceEngine(
    context: Context,
    modelPath: String
) {
    private val interpreter: Interpreter
    private val tokenizer: Tokenizer
    
    init {
        val modelFile = loadModelFile(context, modelPath)
        val options = Interpreter.Options().apply {
            setNumThreads(4)
        }
        interpreter = Interpreter(modelFile, options)
        tokenizer = loadTokenizer(context, "t5_tokenizer.json")
    }
    
    fun summarize(
        text: String,
        maxLength: Int,
        minLength: Int
    ): String {
        // 1. Tokenization
        val inputIds = tokenizer.encode("summarize: $text", maxLength = 512)
        
        // 2. T5 推理（Encoder-Decoder）
        val decoderInputIds = intArrayOf(0)  // Start token
        val outputIds = mutableListOf<Int>()
        
        repeat(maxLength) {
            val outputs = interpreter.run(
                mapOf(
                    "input_ids" to inputIds,
                    "decoder_input_ids" to decoderInputIds
                )
            )
            
            val logits = outputs["logits"] as FloatArray
            val nextTokenId = argmax(logits)
            
            if (nextTokenId == 1) break  // End token
            outputIds.add(nextTokenId)
        }
        
        // 3. Decode
        return tokenizer.decode(outputIds.toIntArray())
    }
}
```
