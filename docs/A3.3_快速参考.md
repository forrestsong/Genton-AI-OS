### 1.1 ActionExecutorDispatcher

**路径**: `overlay/frameworks/base/services/core/java/com/android/server/genton/ActionExecutorDispatcher.java`

三类执行器从存根改为真实系统调用：

```java
public class ActionExecutorDispatcher {
    public void dispatch(String executorType, String actionId, 
                        String argumentsJson, CapabilityToken token) throws Exception {
        IActionExecutor executor = mExecutors.get(executorType);
        if (executor == null) {
            throw new IllegalArgumentException("Unknown executor type: " + executorType);
        }
        executor.execute(actionId, argumentsJson, token);
    }
    
    // 注册 5 个执行器
    private void registerExecutors() {
        mExecutors.put("input", new InputExecutor());
        mExecutors.put("activity", new ActivityExecutor());
        mExecutors.put("device_control", new DeviceControlExecutor());
        mExecutors.put("query", new QueryExecutor());          // 保留存根
        mExecutors.put("notification", new NotificationExecutor()); // 保留存根
    }
}
```

```java
private void injectTap(int x, int y, boolean longTap, int durationMs) {
    long downTime = SystemClock.uptimeMillis();
    MotionEvent down = MotionEvent.obtain(downTime, downTime,
        MotionEvent.ACTION_DOWN, x, y, 0);
    down.setSource(InputDevice.SOURCE_TOUCHSCREEN);
    injectMotionEvent(down);
    
    if (longTap) {
        Thread.sleep(durationMs);
    }
    
    long upTime = SystemClock.uptimeMillis();
    MotionEvent up = MotionEvent.obtain(downTime, upTime,
        MotionEvent.ACTION_UP, x, y, 0);
    injectMotionEvent(up);
}

private void injectMotionEvent(MotionEvent event) {
    if (!mInputManager.injectInputEvent(event, 
            InputManager.INJECT_INPUT_EVENT_MODE_ASYNC)) {
        throw new RuntimeException("Failed to inject motion event");
    }
    event.recycle();
}
```

#### ui.swipe

10 步分段移动：
```java
case "ui.swipe": {
    int startX = args.getInt("start_x");
    int startY = args.getInt("start_y");
    int endX = args.getInt("end_x");
    int endY = args.getInt("end_y");
    int durationMs = args.optInt("duration_ms", 300);
    
    long downTime = SystemClock.uptimeMillis();
    injectMotionEvent(MotionEvent.obtain(downTime, downTime,
        MotionEvent.ACTION_DOWN, startX, startY, 0));
    
    // 10 步插值
    for (int i = 1; i <= 10; i++) {
        float t = i / 10.0f;
        int x = (int) (startX + (endX - startX) * t);
        int y = (int) (startY + (endY - startY) * t);
        long eventTime = downTime + (durationMs * i / 10);
        injectMotionEvent(MotionEvent.obtain(downTime, eventTime,
            MotionEvent.ACTION_MOVE, x, y, 0));
        Thread.sleep(durationMs / 10);
    }
    
    long upTime = SystemClock.uptimeMillis();
    injectMotionEvent(MotionEvent.obtain(downTime, upTime,
        MotionEvent.ACTION_UP, endX, endY, 0));
}
```

```java
case "app.launch": {
    String packageName = args.getString("package_name");
    String activityName = args.optString("activity_name", "");
    
    Intent intent;
    if (activityName != null && !activityName.isEmpty()) {
        // 显式启动 + 非导出拦截
        ComponentName component = new ComponentName(packageName, activityName);
        if (!isActivityExportedOrWhitelisted(component)) {
            throw new SecurityException(
                "Cannot launch non-exported activity: " + 
                component.flattenToShortString() + 
                " (not in system whitelist)");
        }
        intent = new Intent();
        intent.setComponent(component);
    } else {
        // 隐式启动（launcher intent）
        intent = mPackageManager.getLaunchIntentForPackage(packageName);
        if (intent == null) {
            throw new RuntimeException("No launch intent for: " + packageName);
        }
    }
    
    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
    mContext.startActivity(intent);
}
```

**非导出组件 fail-secure 拦截**:
```java
private boolean isActivityExportedOrWhitelisted(ComponentName component) {
    try {
        ActivityInfo info = mPackageManager.getActivityInfo(component, 0);
        if (info.exported) {
            return true;
        }
        // 系统组件白名单回退
        return mSystemActivityWhitelist.contains(component.flattenToShortString());
    } catch (PackageManager.NameNotFoundException e) {
        return false;
    }
}
```

```java
case "app.foreground": {
    String packageName = args.getString("package_name");
    
    // 方案1: 遍历任务栈 + moveTaskToFront
    List<ActivityManager.RunningTaskInfo> tasks = 
        mActivityManager.getRunningTasks(Integer.MAX_VALUE);
    for (ActivityManager.RunningTaskInfo task : tasks) {
        if (task.baseIntent.getComponent().getPackageName().equals(packageName)) {
            mActivityManager.moveTaskToFront(task.id, 0);
            return;
        }
    }
    
    // 方案2: REORDER_TO_FRONT 回退
    Intent intent = mPackageManager.getLaunchIntentForPackage(packageName);
    intent.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | 
                    Intent.FLAG_ACTIVITY_REORDER_TO_FRONT);
    mContext.startActivity(intent);
}
```

### 3.3 白名单配置

**加载逻辑**:
```java
private Set<String> loadSystemActivityWhitelist() {
    Set<String> whitelist = new HashSet<>();
    File whitelistFile = new File("/vendor/genton/config/system_activity_whitelist.json");
    
    if (whitelistFile.exists()) {
        try {
            String content = new String(Files.readAllBytes(whitelistFile.toPath()));
            JSONObject json = new JSONObject(content);
            JSONArray array = json.getJSONArray("whitelist");
            for (int i = 0; i < array.length(); i++) {
                whitelist.add(array.getString(i));
            }
        } catch (Exception e) {
            Slog.e(TAG, "Failed to load whitelist, using defaults", e);
        }
    }
    
    // 内置5个兜底
    if (whitelist.isEmpty()) {
        whitelist.add("com.android.settings/.SubSettings");
        whitelist.add("com.android.settings/.wifi.WifiSettings");
        whitelist.add("com.android.settings/.bluetooth.BluetoothSettings");
        whitelist.add("com.android.settings/.DisplaySettings");
        whitelist.add("com.android.settings/.SoundSettings");
    }
    
    return whitelist;
}
```

**配置文件**: `vendor/genton/config/system_activity_whitelist.json`（16个系统组件）

```java
case "device.volume": {
    String streamName = args.getString("stream");
    String direction = args.getString("direction");
    
    int streamType = getStreamType(streamName);
    int adjustDirection;
    switch (direction) {
        case "up":     adjustDirection = AudioManager.ADJUST_RAISE; break;
        case "down":   adjustDirection = AudioManager.ADJUST_LOWER; break;
        case "mute":   adjustDirection = AudioManager.ADJUST_MUTE; break;
        case "unmute": adjustDirection = AudioManager.ADJUST_UNMUTE; break;
        default: throw new IllegalArgumentException("Unknown direction: " + direction);
    }
    
    mAudioManager.adjustStreamVolume(streamType, adjustDirection, 
        AudioManager.FLAG_SHOW_UI);
    
    int newVolume = mAudioManager.getStreamVolume(streamType);
}

private int getStreamType(String streamName) {
    switch (streamName.toLowerCase()) {
        case "media":        return AudioManager.STREAM_MUSIC;
        case "ring":         return AudioManager.STREAM_RING;
        case "alarm":        return AudioManager.STREAM_ALARM;
        case "notification": return AudioManager.STREAM_NOTIFICATION;
        case "voice_call":   return AudioManager.STREAM_VOICE_CALL;
        case "system":       return AudioManager.STREAM_SYSTEM;
        default: throw new IllegalArgumentException("Unknown stream: " + streamName);
    }
}
```
